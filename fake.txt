local function PerformInventoryScan()
    if isScanning then return end
    isScanning = true
    getgenv().UGCs = {}
    
    tweenService:Create(scanBtn, TweenInfo.new(0.15), {
        BackgroundColor3 = Color3.fromRGB(20, 20, 22),
        TextColor3 = Color3.fromRGB(100, 100, 105)
    }):Play()
    scanBtn.Text = "SCANNING PRESET..."

    local LP = Players.LocalPlayer
    local pageIndex = 1
    AddLog("SCRIPT: CHECKING INVENTORY...", Color3.fromRGB(220, 220, 225))

    local privacyUrl = string.format(
        "https://inventory.roblox.com/v2/users/%s/inventory?assetTypes=Gear&limit=10",
        LP.UserId
    )
    local pSuccess, pResponse = pcall(function()
        return request({Url = privacyUrl, Method = "GET"})
    end)
    
    if pSuccess and pResponse.Body then
        if pResponse.StatusCode == 403 or (
            string.find(pResponse.Body, "errors") and
            string.find(pResponse.Body, "is private")
        ) then
            AddLog("SCRIPT: YOUR INVENTORY IS PRIVATE!", Color3.fromRGB(255, 50, 50))
            AddLog("SCRIPT: MAKE YOUR INVENTORY PUBLIC", Color3.fromRGB(255, 120, 120))
            scanBtn.Text = "START FAKE PURCHASE"
            tweenService:Create(scanBtn, TweenInfo.new(0.15), {
                BackgroundColor3 = BTN_NORMAL,
                TextColor3 = Color3.fromRGB(240, 240, 245)
            }):Play()
            isScanning = false
            return false
        end
    end

    local scanTargets = {}
    if selectedCategories.Accessories then
        table.insert(scanTargets, {
            Url = "https://inventory.roblox.com/v2/users/%s/inventory?assetTypes=HairAccessory,FaceAccessory,NeckAccessory,ShoulderAccessory,FrontAccessory,BackAccessory,WaistAccessory,Hat&limit=100&cursor=%s",
            Key = "Accessories"
        })
    end
    if selectedCategories.ClassicClothing then
        table.insert(scanTargets, {
            Url = "https://inventory.roblox.com/v2/users/%s/inventory?assetTypes=Shirt,Pants,TShirt&limit=100&cursor=%s",
            Key = "ClassicClothing"
        })
    end
    if selectedCategories.LayeredClothing then
        table.insert(scanTargets, {
            Url = "https://inventory.roblox.com/v2/users/%s/inventory?assetTypes=JacketAccessory,SweaterAccessory,PantsAccessory,ShirtAccessory&limit=100&cursor=%s",
            Key = "LayeredClothing"
        })
    end
    if selectedCategories.Gears then
        table.insert(scanTargets, {
            Url = "https://inventory.roblox.com/v2/users/%s/inventory?assetTypes=Gear&limit=100&cursor=%s",
            Key = "Gears"
        })
    end
    if selectedCategories.BundlesPackages then
        table.insert(scanTargets, {
            Url = "https://inventory.roblox.com/v1/users/%s/assets/sub-type/AvatarAnimation?limit=100&cursor=%s",
            Key = "BundlesPackages"
        })
        table.insert(scanTargets, {
            Url = "https://inventory.roblox.com/v1/users/%s/assets/sub-type/Outfit?limit=100&cursor=%s",
            Key = "BundlesPackages"
        })
        table.insert(scanTargets, {
            Url = "https://inventory.roblox.com/v1/users/%s/assets/sub-type/Package?limit=100&cursor=%s",
            Key = "BundlesPackages"
        })
        table.insert(scanTargets, {
            Url = "https://catalog.roblox.com/v1/users/%s/bundles?limit=100&cursor=%s",
            Key = "BundlesPackages"
        })
    end

    if #scanTargets == 0 then
        AddLog("SCRIPT: SELECT AT LEAST ONE CATEGORY FIRST!", Color3.fromRGB(255, 100, 100))
        scanBtn.Text = "START FAKE PURCHASE"
        tweenService:Create(scanBtn, TweenInfo.new(0.15), {
            BackgroundColor3 = BTN_NORMAL,
            TextColor3 = Color3.fromRGB(240, 240, 245)
        }):Play()
        isScanning = false
        return false
    end

    for _, target in ipairs(scanTargets) do
        local cursor = ""
        repeat
            local targetUrl = string.format(target.Url, LP.UserId, cursor)
            local success, response = pcall(function()
                return request({Url = targetUrl, Method = "GET"})
            end)

            if success and response.Body then
                local data = HttpService:JSONDecode(response.Body)
                if data.data then
                    local itemsToAnalyze = {}
                    for _, v in pairs(data.data) do
                        local id = v.assetId or v.id
                        if id then
                            table.insert(itemsToAnalyze, {Id = id, Raw = v})
                        end
                    end

                    for _, item in ipairs(itemsToAnalyze) do
                        local keepItem = true
                        if skipFreeItems then
                            local infoSuccess, info = pcall(function()
                                return MarketplaceService:GetProductInfo(
                                    item.Id,
                                    target.Key == "BundlesPackages" and Enum.InfoType.Bundle or Enum.InfoType.Asset
                                )
                            end)
                            
                            if infoSuccess and info then
                                local isFree = (
                                    info.PriceInRobux == 0 or
                                    info.IsForSale == false or
                                    info.CanBePurchased == false
                                )
                                if isFree then
                                    keepItem = false
                                end
                            else
                                keepItem = false
                            end
                        end

                        if keepItem then
                            table.insert(getgenv().UGCs, {Id = item.Id, Type = target.Key})
                        end
                    end

                    AddLog("SCANNING: PAGE " .. pageIndex .. " SCANNED SUCCESSFULLY", Color3.fromRGB(0, 200, 255))
                    pageIndex = pageIndex + 1
                end
                cursor = data.nextPageCursor or ""
            else
                task.wait(0.5)
            end
            task.wait(0.1)
        until cursor == ""
    end

    AddLog("SCRIPT: TOTAL ITEMS SCANNED: " .. #getgenv().UGCs, Color3.fromRGB(0, 255, 0))
    scanBtn.Text = "START FAKE PURCHASE"
    tweenService:Create(scanBtn, TweenInfo.new(0.15), {
        BackgroundColor3 = BTN_NORMAL,
        TextColor3 = Color3.fromRGB(240, 240, 245)
    }):Play()
    isScanning = false
    isLoopRunning = true
    return true
end

task.spawn(function()
    local LP = Players.LocalPlayer
    local loopPageIndex = 1
    AddLog("SCRIPT: FAKE PURCHASER...", Color3.fromRGB(255, 255, 0))
    AddLog("SCRIPT: SELECT ANY CATEGORIES YOU WANT FIRE", Color3.fromRGB(200, 200, 205))

    while true do
        if isLoopRunning then
            local queueSize = #getgenv().UGCs
            if queueSize > 0 then
                for i = 1, queueSize do
                    if not isLoopRunning then break end
                    
                    local itemData = getgenv().UGCs[i]
                    if not itemData then continue end
                    
                    if not selectedCategories[itemData.Type] then
                        continue
                    end
                    
                    local success, err = pcall(function()
                        MarketplaceService:SignalPromptPurchaseFinished(LP, itemData.Id, true)
                    end)

                    if success then
                        AddLog("SCRIPT: ITEM ID FIRED " .. itemData.Id, Color3.fromRGB(150, 255, 150))
                    else
                        AddLog("SCRIPT: ITEM ID BLOCKED " .. itemData.Id, Color3.fromRGB(255, 100, 100))
                    end
                    
                    if fireSpeed > 0 then
                        task.wait(fireSpeed)
                    else
                        runService.Heartbeat:Wait()
                    end
                end
                if isLoopRunning then
                    AddLog("SCRIPT: SCANNED SUCCESSFULLY", Color3.fromRGB(200, 200, 200))
                    isLoopRunning = false
                end
            else
                task.wait(1)
            end
        else
            task.wait(0.5)
        end
    end
end)

do
    local dragging, dragStart, startPos = false, nil, nil
    main.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            local abs = main.AbsolutePosition
            main.Position = UDim2.new(0, abs.X, 0, abs.Y)
            dragging = true
            dragStart = i.Position
            startPos = main.Position
        end
    end)
    
    userInputService.InputChanged:Connect(function(i)
        if not dragging then return end
        if i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch then
            local d = i.Position - dragStart
            local vp = workspace.CurrentCamera.ViewportSize
            local gs = main.AbsoluteSize
            main.Position = UDim2.new(
                0,
                math.clamp(startPos.X.Offset + d.X, 0, vp.X - gs.X),
                0,
                math.clamp(startPos.Y.Offset + d.Y, 0, vp.Y - gs.Y)
            )
        end
    end)
    
    userInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

    local off = Vector2.new(math.sin(tick() * 2.8), 0)
    for _, g in ipairs(allGrads) do
        g.Offset = off
    end
end)
